# 内容规格 JSON Schema（spec.json）

生成器输入。完整可运行示例见 [examples/nxm_spec.json](../examples/nxm_spec.json)。
缺失键会导致生成失败或校验失败；`risks` / `cross_validation` / `conditional_approval` 可选（提供才生成附录页）。

```jsonc
{
  // ── 标题区 ──
  "title_category": "改良改进",            // 改良改进 | 全新开发 | 定制开发
  "title_product": "NXM系列石化行业专供塑壳断路器开发",
  "org_unit": "低压电器研究院塑壳研发部",
  "team": {                                // 7 角色必填（可为空串但要给键）
    "负责人": "王工程师", "产品": "李经理", "研发": "张工", "制造": "刘工",
    "质量": "陈工", "财务": "赵经理", "供应链": "孙经理"
  },

  // ── 顶部标签牌（平台只允许 昆仑/新锐/诺雅克）──
  "meta": { "需求来源": "中国区", "规划方向": "高性价比", "产品平台": "新锐" },

  // ── 项目计划 ──
  "plan": { "phases": [
    { "name": "P1（项目输入与评估）", "time": "2026.05" },
    { "name": "P2（设计验证）",       "time": "2026.05-06" },
    { "name": "P3（可生产准备与验证）", "time": "2026.06-07" },
    { "name": "P4（可销售准备）",     "time": "2026.07-08" }
  ]},

  // ── 左列 ──
  "research": "调研时间+对象数量+规格区间+现有产品缺口+通路限定（数字化表述）",
  "scenario_pain": {
    "应用场景": "1-2 句工况描述",
    "客户痛点": [              // 每条必须量化：要求值 vs 现有值
      "高温降容：70℃要求降容≤10%，现有产品降容15%"
    ]
  },
  "quick_facts": { "竞争对手": "常熟CM3（盘厂供货价约18折）", "目标行业": "石化", "销售通路": "OEM + 渠道" },

  // ── 中列 ──
  "product_definition": {
    "外观": "平台/配色/标识", "材质": "关键材料", "认证": "CCC/CE",
    "性能参数": [ "逐条回应客户痛点，含单位数值" ]
  },
  "price_cost": {
    "策略": "面价 vs 竞品 %、目标成本、材料成本占比、毛利率",
    "spec_table": {
      "headers": ["规格", "占比", "口袋价（含税）", "成本（未税）", "毛利率"],
      "rows": [["3P 630", "25%", "3,800", "2,210", "40%"]],
      "col_widths": [1.1, 0.8, 1.4, 1.3, 1.0]     // 可选，相对宽度
    }
  },

  // ── 右列 ──
  "technical_feasibility": {
    "verdict": "部分可行",        // 可行 | 部分可行 | 不可行（标题条后缀，部分可行标红）
    "方案引言": "项目总体方案：在XX平台基础上改进，方案如下：",
    "方案": [ "1、…；", "2、…；", "3、…；" ],
    "创新点": "…",
    "不满足项": "红色警示行；verdict=可行时可省略"    // 部分可行时必填并指向附录A
  },
  "production_feasibility": {
    "verdict": "可行",
    "人员": "…", "设备": "…", "场地": "…", "物料": "…"
  },
  "finance": {
    "summary": "项目投入合计X万元；预计Y年收回投资；盈亏平衡销售额Z万元",
    "investment": {
      "headers": ["研发材料费", "研发试制费", "研发人工费", "设备设施", "其他", "合计"],
      "rows": [["1.5", "0.8", "2.4", "0", "3.5", "8.2"]]   // 单位：万元
    },
    "forecast_title": "销售预测：",
    "forecast": {
      "headers": ["", "第一年", "第二年", "第三年", "三年合计"],
      "rows": [
        ["数量（台）", "800", "850", "900", "2,550"],
        ["销售额（万元）", "304", "323", "342", "969"],
        ["毛利润（万元）", "118.6", "126.0", "133.4", "378.0"]
      ]
    }
  },

  // ── 附录A：风险预案（有风险必给，四要素齐全）──
  "risks": [{
    "风险": "…", "影响": "…", "预案": "…（含量化措施）",
    "资源诉求": "…", "责任人": "…"
  }],
  "conditional_approval": ["有条件立项条件 1", "…"],

  // ── 附录B：销售预测交叉验证 ──
  "cross_validation": {
    "steps": [
      { "步骤": "行业总出货量", "推算": "…（标注来源）", "数据来源": "电器工业协会年报" }
    ],
    "对比结论": "加粗结论：偏差是否合理、预测是否可信"
  }
}
```

## 生成与校验

```bash
python3 scripts/generate_one_pager.py spec.json out.pptx
python3 scripts/validate_one_pager.py out.pptx --spec spec.json   # PASS 才可交付
```
